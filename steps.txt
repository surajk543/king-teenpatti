# Deploy routine — Go server (the Node server was removed from production on the switch)
cd /var/www/gameplay/king-teenpatti
git pull origin master
bash go-server/ops/build.sh                       # as deploy, no sudo: installs Go if missing, builds go-server/bin/gameplay
sudo bash go-server/ops/install-go-server.sh      # FIRST TIME ONLY: swaps gameplay.service to the Go binary, copies .env, removes server/
sudo systemctl restart gameplay                   # every later deploy (after build.sh)
sudo systemctl status gameplay --no-pager
sudo journalctl -u gameplay -n 20 --no-pager
curl -s http://127.0.0.1:3000/health | python3 -m json.tool | head -20   # process.node must start with "go"

# Which build is live? /health carries the release tag ops/build.sh stamped in.
# Do this from anywhere — no ssh, no credentials — and trust nothing until it says IN SYNC:
# a restart that quietly failed is indistinguishable from a successful one otherwise.
bash go-server/ops/prod-version.sh                # IN SYNC / BEHIND (exit 2) against api.sungamestudio.com
curl -s https://api.sungamestudio.com/health | python3 -c 'import sys,json;print(json.load(sys.stdin)["version"])'

# Cutting a release (tags are go-server/vX.Y.Z; the script refuses a dirty tree and never pushes):
bash go-server/ops/release.sh patch               # or minor / major / vX.Y.Z ; --current just reports
git push origin go-server/vX.Y.Z
