# Deploy routine — Go server. Production runs a TAG (go-server/vX.Y.Z), never "whatever master is now".

# 0. FIRST, on your own machine: merge the release PR, then cut the tag ON master and push it.
#    The tag IS the version — build.sh stamps it into the binary and /health reports it — so an
#    untagged deploy has no name. Cut it on master AFTER the merge, on the merge commit itself: a tag
#    on the PR branch's last commit is one commit behind what master holds, and a host building master
#    would report vX.Y.Z-1-g<sha>. release.sh refuses local edits (and untracked files under
#    go-server/, which would be compiled in), a commit that is already tagged, and never pushes.
git checkout master && git pull origin master
bash go-server/ops/release.sh patch               # or minor / major / vX.Y.Z ; --current just reports
git push origin go-server/vX.Y.Z

# 1. Then on the host, as the checkout's owner (write) — ONE command since 29 Sep 2026 (DEPLOY.md §3 "One-command deploy"),
#    INSIDE tmux (or screen): a dropped ssh session must not end a deploy half way.
tmux new -A -s deploy                                                       # after a drop: tmux attach -t deploy
bash /var/www/gameplay/king-teenpatti/go-server/ops/deploy.sh --dry-run     # what it would do; changes nothing
bash /var/www/gameplay/king-teenpatti/go-server/ops/deploy.sh               # the newest go-server/v* tag (or name one: … deploy.sh go-server/vX.Y.Z)
#   It fetches (branches + tags), fast-forwards master, checks the tag out DETACHED, builds it, runs the NEW binary's
#   `gameplay -migrate` as the service's user BEFORE the restart (a failing script stops it with nothing restarted),
#   restarts gameplay, waits for /health to report the tag, watches it 30 s more, and rolls back by itself when it
#   never comes up or falls over. Every run is logged in .git/deploy-logs/ (it names the file).
#   Never `sudo bash …` it (it asks sudo itself — and may ask again after a long first build), never `psql -f` the
#   migration files as postgres (DEPLOY.md says why), and no more `git pull origin master` on the host: the checkout is
#   left detached at the tag, and the script fetches.
#   A ROLLBACK is a deploy of the older tag with --allow-downgrade, after reading DEPLOY.md §5; it refuses an older tag
#   without it, and any tag older than go-server/v1.7.0 always (this database cannot run one).
#   Deploy a release whose migrations change the structure (the dry run lists the changed scripts) at a quiet hour:
#   while its new DDL waits for a lock (3 s at most), the writes to that table wait behind it.
#   FIRST TIME the script is on the host — it arrives with the first checkout that holds it. The host sits on a
#   detached HEAD at the tag it runs (since the v1.10.2 deploy), so fetch and check master out detached, then run it:
#     cd /var/www/gameplay/king-teenpatti && git fetch origin && git checkout --detach origin/master && bash go-server/ops/deploy.sh
#
# By hand (what the script does, without its checks; and the one-time unit install):
cd /var/www/gameplay/king-teenpatti
TAG=go-server/vX.Y.Z                              # the tag to run (an older one rolls back — read DEPLOY.md §5 first)
git fetch origin --tags --force                   # brings the tag (--force: also one that was moved)
git status --short --untracked-files=no           # must print nothing — a local edit would be built in
git checkout --detach "$TAG"                      # the checkout now IS the tag (a detached HEAD, on purpose)
bash go-server/ops/build.sh                       # as the checkout's owner, no sudo: installs Go if missing, builds go-server/bin/gameplay
./go-server/bin/gameplay -version                 # must name the tag: "gameplay vX.Y.Z go1.27.1 linux/amd64"
sudo bash go-server/ops/install-go-server.sh      # FIRST TIME ONLY: swaps gameplay.service to the Go binary, copies .env, removes server/
(cd go-server && sudo -u "$(systemctl show gameplay -p User --value)" env NODE_ENV=production ./bin/gameplay -migrate) && sudo systemctl restart gameplay
#   ^ the tag's migrations first (as the unit's User= — `gameplay` on production, read from the unit — with its NODE_ENV,
#     as deploy.sh does), and the restart ONLY if
#     they succeeded: one line, so a failed migration leaves the old build serving even in a pasted block
sudo systemctl status gameplay --no-pager
sudo journalctl -u gameplay -n 20 --no-pager
for i in $(seq 60); do curl -sf -o /dev/null http://127.0.0.1:3000/health && break; sleep 1; done   # the restart first settles live pots (up to 20s), then boots
curl -s http://127.0.0.1:3000/health | python3 -c 'import sys,json;d=json.load(sys.stdin);print(d["ok"], d["version"])'   # True vX.Y.Z
#    Roll back = deploy.sh go-server/vPREVIOUS --allow-downgrade (after it, a tag older than deploy.sh leaves no deploy.sh:
#    the next deploy is the FIRST TIME line above again), or these lines with that tag (list them:
#    git tag -l 'go-server/v*' --sort=-v:refname | head -5). The host stays on a detached HEAD at the tag it
#    runs; the next deploy's fetch + checkout moves it on. Do not `git pull origin master` there any more:
#    master can be ahead of the last tag, and a pull changes the source the NEXT build would compile without
#    changing what is running.

# The host's go-server/.env OVERRIDES the defaults compiled into the binary. A release that changes
# a default (LOBBY_TABLES, TABLE_STAKES, SEEN_MAX_POT, …) does nothing on a host that pins that key
# until the file is edited — check it whenever the release notes mention one.
grep -E '^(LOBBY_TABLES|TABLE_STAKES|SEEN_MAX_POT|ENTRY_CAP|TABLE_CONFIG_SOURCE)' go-server/.env
# Since 23 Sep 2026 the tables can come from PostgreSQL instead (TABLE_CONFIG_SOURCE=db; DEPLOY.md §3):
# then those keys are ignored, a release's new default table does not reach the existing rows either
# (the seed adds a new one INACTIVE), and an edit is an UPDATE + this restart. Which one is playing:
curl -s http://127.0.0.1:3000/health | python3 -c 'import sys,json;print(json.load(sys.stdin)["tableConfig"])'   # source env|db, fallback must be False

# Which build is live? /health carries the release tag ops/build.sh stamped in.
# Do this from anywhere — no ssh, no credentials — and trust nothing until it says IN SYNC:
# a restart that quietly failed is indistinguishable from a successful one otherwise.
bash go-server/ops/prod-version.sh                # IN SYNC when prod runs the newest tag; BEHIND (exit 2) otherwise — expected after a deliberate rollback
curl -s https://prod.sungamestudio.com/health | python3 -c 'import sys,json;print(json.load(sys.stdin)["version"])'
